Porkbun Batch Domain Price Checker

CRITICAL: epic.lol and boss.mom were found to be PREMIUM domains at $2,183.84/yr despite showing as "available" in DNS/WHOIS. Many registries mark common dictionary words as premiums. Always verify actual pricing before buying.

How to Use

  1. Go to porkbun.com and log in
  2. Open browser DevTools (Cmd+Option+J on Mac)
  3. Paste the script below into the Console tab
  4. Press Enter — it will check each domain and show real prices
  5. Results appear in the console as a table

Browser Console Script

// === PORKBUN BATCH PRICE CHECKER ===
// Paste this in your browser console at porkbun.com
// It uses Porkbun's internal search API (same session auth)

const DOMAINS = [
  // Hair ($1.54 base)
  'silk.hair','glow.hair','pure.hair','bold.hair','soft.hair','gold.hair',
  'wave.hair','curl.hair','jade.hair','mint.hair','rose.hair','lush.hair',
  'wild.hair','long.hair','flex.hair',
  // Skin ($1.54 base)
  'glow.skin','pure.skin','silk.skin','soft.skin','gold.skin','bare.skin',
  'rose.skin','mist.skin','mint.skin','dawn.skin',
  // Beauty ($1.54 base)
  'ruby.beauty','pure.beauty','bold.beauty','glow.beauty','dawn.beauty',
  'wild.beauty','lush.beauty','silk.beauty','rose.beauty','soft.beauty',
  // Quest ($1.54 base)
  'ghost.quest','wolf.quest','sus.quest',
  // LOL ($1.54 base)
  'epic.lol','chad.lol','wild.lol',
  // Monster ($1.54 base)
  'your.monster','chad.monster','foam.monster',
  // Click ($1.54 base)
  'do.click','go.click','fork.click','dash.click',
  // Beer ($1.54 base)
  'hop.beer','hops.beer','pine.beer','lime.beer','plum.beer',
  // Mom ($1.54 base)
  'boss.mom','best.mom','play.mom','wise.mom',
  // Rest/Pics/Garden ($1.54 base)
  'good.rest','star.pics','cool.pics','dew.garden',
  // Surf ($1.54 base)
  'tide.surf','dawn.surf','dusk.surf','mist.surf','gold.surf',
  // Best ($1.72 base)
  'stay.best','win.best','code.best','data.best','cook.best',
  // Space ($1.96 base)
  'hack.space','zen.space','dawn.space','mist.space',
];

const results = [];
const DELAY_MS = 2000; // 2 sec between checks (be nice)

async function checkDomain(domain) {
  try {
    const resp = await fetch(`/checkout/api/checkDomain?search=${encodeURIComponent(domain)}`, {
      method: 'GET',
      credentials: 'same-origin',
    });
    const data = await resp.json();
    return data;
  } catch (e) {
    return { error: e.message };
  }
}

async function sleep(ms) { return new Promise(r => setTimeout(r, ms)); }

console.log(`%c=== PORKBUN BATCH PRICE CHECK: ${DOMAINS.length} domains ===`, 'color: #ff6b9d; font-size: 16px');

for (let i = 0; i < DOMAINS.length; i++) {
  const domain = DOMAINS[i];
  const data = await checkDomain(domain);
  
  let price = '?';
  let renew = '?';
  let isPremium = false;
  let status = 'unknown';
  
  // Try to extract pricing from response
  if (data && data.pricing) {
    price = data.pricing.registration || '?';
    renew = data.pricing.renewal || '?';
    isPremium = data.premium === true || parseFloat(price) > 50;
    status = data.avail === 'yes' ? 'available' : 'taken';
  } else if (data && data.domains) {
    // Alternative response format
    const match = data.domains?.find(d => d.domain === domain);
    if (match) {
      price = match.price || match.registration || '?';
      renew = match.renewal || '?';
      isPremium = match.premium || parseFloat(price) > 50;
      status = match.available ? 'available' : 'taken';
    }
  }
  
  const result = { domain, price, renew, isPremium, status, raw: data };
  results.push(result);
  
  if (isPremium) {
    console.log(`%c  PREMIUM: ${domain} — $${price}/yr (renews $${renew}/yr) SKIP!`, 'color: #ff4444; font-weight: bold');
  } else if (status === 'available') {
    console.log(`%c  OK: ${domain} — $${price}/yr (renews $${renew}/yr)`, 'color: #00ff88');
  } else {
    console.log(`  ${status}: ${domain} — $${price}/yr`, '');
  }
  
  // Progress
  if ((i + 1) % 10 === 0) {
    console.log(`%c  ... ${i + 1}/${DOMAINS.length} checked ...`, 'color: #888');
  }
  
  if (i < DOMAINS.length - 1) await sleep(DELAY_MS);
}

// Summary
const affordable = results.filter(r => !r.isPremium && r.status === 'available');
const premiums = results.filter(r => r.isPremium);
const taken = results.filter(r => r.status === 'taken');

console.log('\n%c=== RESULTS ===', 'color: #ff6b9d; font-size: 16px');
console.log(`%c  Affordable + Available: ${affordable.length}`, 'color: #00ff88; font-size: 14px');
console.log(`%c  PREMIUM (skip): ${premiums.length}`, 'color: #ff4444; font-size: 14px');
console.log(`  Taken: ${taken.length}`);

if (affordable.length > 0) {
  console.log('\n%c  BUY THESE:', 'color: #00ff88; font-weight: bold');
  console.table(affordable.map(r => ({
    Domain: r.domain,
    Price: `$${r.price}/yr`,
    Renewal: `$${r.renew}/yr`,
  })));
}

if (premiums.length > 0) {
  console.log('\n%c  PREMIUM — DO NOT BUY:', 'color: #ff4444; font-weight: bold');
  console.table(premiums.map(r => ({
    Domain: r.domain,
    Price: `$${r.price}/yr`,
    Renewal: `$${r.renew}/yr`,
  })));
}

// Save results to clipboard
const report = results.map(r => `${r.domain}\t${r.status}\t$${r.price}\t$${r.renew}\t${r.isPremium ? 'PREMIUM' : 'ok'}`).join('\n');
navigator.clipboard.writeText(report).then(() => {
  console.log('%c  Results copied to clipboard (tab-separated)', 'color: #88ccff');
});
